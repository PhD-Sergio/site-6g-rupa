{{- /* A responsive WebP from an image resource. Params: img image resource (required) widths candidate widths in px, default (slice 320 640 960); never upscaled sizes the sizes attribute, default "100vw" alt alt text, default "" class optional class eager true to load immediately instead of lazily */ -}} {{- $img := .img -}} {{- $widths := .widths | default (slice 320 640 960) -}} {{- $srcset := slice -}} {{- $fallback := "" -}} {{- range $widths -}} {{- $w := math.Min . $img.Width | int -}} {{- $r := $img.Resize (printf "%dx webp q80" $w) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $r.Width) -}} {{- if not $fallback }}{{ $fallback = $r }}{{ end -}} {{- end -}}
